PROGRAM delivery quote

RECORD Parcel
  weight_grams: integer
  member: boolean

RECORD Quote
  status: "standard" | "manual"
  fee: integer

REQUEST quote delivery
  GIVEN parcel is Parcel
  GIVEN quote is Quote
    status: "standard"
    fee: 0
  WHEN quote delivery for parcel into quote
  OUTPUT quote is Quote
  THEN quote.fee >= 0

WHEN quote delivery for <parcel> into <quote>
  GIVEN parcel is Parcel
  AND quote is Quote
  REQUIRE parcel.weight_grams > 0
  DECIDE
    WHEN parcel.weight_grams > 5000
      set quote.status to "manual"
      set quote.fee to 0
    WHEN parcel.member
      set quote.status to "standard"
      set quote.fee to 0
    WHEN parcel.weight_grams <= 1000
      set quote.status to "standard"
      set quote.fee to 5
    ELSE
      set quote.status to "standard"
      set quote.fee to 9

BACKGROUND
GIVEN quote is Quote
  status: "standard"
  fee: 0

SCENARIO weight and membership determine the quote
GIVEN parcel is Parcel
  weight_grams: <weight>
  member: <member>

WHEN quote delivery for parcel into quote

THEN quote.status == <status>
AND quote.fee == <fee>
